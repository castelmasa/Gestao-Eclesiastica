                        <Input id="endereco" name="endereco" defaultValue={editingMember?.endereco} placeholder="Rua das Flores, 123" />
                      </div>
                      <div className="grid gap-2">
                        <Label htmlFor="cep">CEP</Label>
                        <Input id="cep" name="cep" defaultValue={editingMember?.cep} placeholder="00000-000" onChange={handleCepChange} />
                      </div>
                    </div>
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                      <div className="grid gap-2">
                        <Label htmlFor="bairro">Bairro</Label>
                        <Input id="bairro" name="bairro" defaultValue={editingMember?.bairro} placeholder="Centro" />
                      </div>
                      <div className="grid gap-2">
                        <Label htmlFor="cidade">Cidade</Label>
                        <Input id="cidade" name="cidade" defaultValue={editingMember?.cidade} placeholder="São Paulo" />
                      </div>
                    </div>
                  </div>

                  {/* Eclesiástico e Foto */}
                  <div className="space-y-4">
                    <h3 className="text-sm font-medium text-slate-500 border-b pb-2">Dados Eclesiásticos e Foto</h3>
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                      <div className="grid gap-2">
                        <Label htmlFor="role">Cargo / Função</Label>
                        <Input id="role" name="role" defaultValue={editingMember?.role} placeholder="Ex: Membro, Diácono" required />
                      </div>
                      <div className="grid gap-2">
                                              <div className="grid gap-2">
                        <Label htmlFor="tags">Tags (separadas por vírgula)</Label>
                        <Input id="tags" name="tags" defaultValue={editingMember?.tags ? editingMember.tags.join(", ") : ""} placeholder="Ex: Músico, Obreiro, Voluntário" />
                      </div>
                      </div>
                      <div className="grid gap-2">
                        <Label htmlFor="status">Status</Label>
                        <Select name="status" defaultValue={editingMember?.status || "Ativo"} required>
                          <SelectTrigger>
                            <SelectValue placeholder="Selecione..." />
                          </SelectTrigger>
                          <SelectContent>
                            <SelectItem value="Ativo">Ativo</SelectItem>
                            <SelectItem value="Inativo">Inativo</SelectItem>
                          </SelectContent>
                        </Select>
                      </div>
                    </div>
                    <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
                      <div className="grid gap-2">
                        <Label htmlFor="dataBatismo">Data do Batismo</Label>
                        <Input id="dataBatismo" name="dataBatismo" type="date" defaultValue={parseBRDateForInput(editingMember?.dataBatismo)} />
                      </div>
                      <div className="grid gap-2">
                        <Label htmlFor="membroDesde">Membro Desde</Label>
                        <Input id="membroDesde" name="membroDesde" type="date" defaultValue={parseBRDateForInput(editingMember?.membroDesde)} />
                      </div>
                      <div className="grid gap-2">
                        <Label htmlFor="validade">Validade (Carteirinha)</Label>
                        <Input id="validade" name="validade" type="date" defaultValue={parseBRDateForInput(editingMember?.validade)} />
                      </div>
                    </div>
                    <div className="grid gap-2">
                      <Label htmlFor="photo">Upload de Foto (Para Carteirinha)</Label>
                      {photoPreview && (
                        <div className="mt-2 mb-2">
                          <img src={photoPreview} alt="Preview da Foto" className="w-24 h-24 object-cover rounded-md border border-border shadow-sm" />
                        </div>
                      )}
                      <Input 
                        id="photo" 
                        name="photo" 
                        type="file" 
                        accept="image/*" 
                        onChange={(e) => {
                          const file = e.target.files?.[0];
                          if (file) {
                            const reader = new FileReader();
                            reader.onloadend = () => {
                              setPhotoPreview(reader.result as string);
                            };
                            reader.readAsDataURL(file);
                          } else {
                            // Se desmarcar, volta pra original ou nulo
                            setPhotoPreview(editingMember?.photoUrl || null);
                          }
                        }}
                      />
                    </div>
                  </div>

                </div>

                {/* INÍCIO DO HISTÓRICO DE COMUNICAÇÕES */}
                {editingMember && (
                  <div className="col-span-1 sm:col-span-2 pt-4 border-t mt-4">
                    <h3 className="text-lg font-semibold mb-3">Histórico de Comunicações / Cuidado Pastoral</h3>
                    <div className="space-y-4">
                      {editingMember.communications && editingMember.communications.length > 0 ? (
                        <div className="space-y-2 max-h-[200px] overflow-y-auto pr-2">
                          {editingMember.communications.map((comm: any, idx: number) => (
                            <div key={comm.id || idx} className="p-3 bg-muted/30 border rounded-md text-sm">
                              <div className="flex justify-between font-semibold mb-1">
                                <span>{comm.type}</span>
                                <span className="text-muted-foreground">{new Date(comm.date).toLocaleDateString('pt-BR')}</span>
                              </div>
                              <p className="text-muted-foreground">{comm.notes}</p>
                            </div>
                          ))}
                        </div>
                      ) : (
                        <p className="text-sm text-muted-foreground">Nenhuma comunicação registrada ainda.</p>
                      )}
                      
                      <div className="bg-card border rounded-md p-4 space-y-3">
                        <h4 className="text-sm font-semibold">Adicionar Novo Registro</h4>
                        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                           <div className="space-y-1">
                             <Label>Data</Label>
                             <Input type="date" id="newCommDate" defaultValue={new Date().toISOString().split('T')[0]} />
                           </div>
                           <div className="space-y-1">
                             <Label>Tipo de Contato</Label>
                             <Select defaultValue="Visita Pastoral">
                               <SelectTrigger id="newCommType"><SelectValue /></SelectTrigger>
                               <SelectContent>
                                 <SelectItem value="Visita Pastoral">Visita Pastoral</SelectItem>
                                 <SelectItem value="Aconselhamento">Aconselhamento</SelectItem>
                                 <SelectItem value="Ligação">Ligação</SelectItem>
                                 <SelectItem value="WhatsApp">WhatsApp</SelectItem>
                                 <SelectItem value="E-mail">E-mail</SelectItem>
                                 <SelectItem value="Outro">Outro</SelectItem>
                               </SelectContent>
                             </Select>
                           </div>
                           <div className="col-span-1 sm:col-span-2 space-y-1">
                             <Label>Observações</Label>
                             <Textarea id="newCommNotes" placeholder="Descreva os detalhes da interação..." rows={2}></Textarea>
                           </div>
                           <div className="col-span-1 sm:col-span-2">
                             <Button type="button" variant="secondary" onClick={() => {
                                const dateVal = (document.getElementById('newCommDate') as HTMLInputElement).value;
                                const typeVal = (document.getElementById('newCommType') as HTMLButtonElement)?.textContent || "Visita Pastoral";
                                const notesVal = (document.getElementById('newCommNotes') as HTMLTextAreaElement).value;
                                
                                if (!notesVal) {
                                  toast.error("Preencha as observações.");
                                  return;
                                }
                                
                                const newComm = { id: Date.now().toString(), date: dateVal, type: typeVal, notes: notesVal };
                                
                                // Update local state directly to reflect
                                setMembers((prev: any) => prev.map((m: any) => {
                                  if (m.id === editingMember.id) {
                                     const updatedComms = m.communications ? [...m.communications, newComm] : [newComm];
                                     const updatedM = { ...m, communications: updatedComms };
                                     setEditingMember(updatedM); // to trigger re-render of this section
                                     return updatedM;
                                  }
                                  return m;
                                }));
                                
                                toast.success("Registro adicionado! Não se esqueça de Salvar as Alterações.");
                                (document.getElementById('newCommNotes') as HTMLTextAreaElement).value = '';
                             }}>Adicionar Registro</Button>
                           </div>
                        </div>
                      </div>
                    </div>
                  </div>
                )}
                {/* FIM DO HISTÓRICO DE COMUNICAÇÕES */}

                <DialogFooter>
                  <Button type="button" variant="outline" onClick={() => setIsMemberModalOpen(false)}>Cancelar</Button>
                  <Button type="submit">{editingMember ? "Salvar Alterações" : "Salvar Membro"}</Button>
                </DialogFooter>
              </form>
            </DialogContent>
          </Dialog>
        </div>
      </div>
